cmake_minimum_required(VERSION 3.22)
project(opensekai_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(opensekai_audio SHARED audio.cpp)
target_include_directories(opensekai_audio PRIVATE "${ORT_INCLUDE}")
target_link_libraries(opensekai_audio PRIVATE "${ORT_LIBRARY}")
if(MSVC)
  target_compile_options(opensekai_audio PRIVATE /O2 /EHsc)
else()
  target_compile_options(opensekai_audio PRIVATE -O3 -fvisibility=hidden)
  target_link_options(opensekai_audio PRIVATE "-Wl,-z,max-page-size=16384")
endif()
